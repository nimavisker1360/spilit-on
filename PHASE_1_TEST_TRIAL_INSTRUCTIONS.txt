---start file---
import { prisma } from "@/lib/prisma";

async function test() {
  const now = new Date();
  const user = await prisma.user.create({
    data: { email: `test-${Date.now()}@test.com`, name: "Test", emailVerified: now }
  });
  const plan = await prisma.subscriptionPlan.findFirst({ where: { code: "trial" } })
    || await prisma.subscriptionPlan.create({
      data: { code: "trial", name: "Trial", monthlyPrice: 0, annualPrice: 0, currency: "TRY",
              includedTables: 10, includedBranches: 1, includedStaff: 5, commissionRate: 0, features: {} }
    });
  const r = await prisma.restaurant.create({
    data: { name: "Test", slug: `test-${Date.now()}`, status: "TRIALING", workspaceMode: "TRIAL",
            currentPlanId: plan.id, trialStartedAt: now, trialEndsAt: new Date(now.getTime() + 14*24*60*60*1000) }
  });
  await prisma.membership.create({ data: { restaurantId: r.id, userId: user.id, role: "OWNER", status: "ACTIVE" } });
  const s = await prisma.tenantSubscription.create({
    data: { restaurantId: r.id, planId: plan.id, status: "TRIALING", billingPeriod: "MONTHLY",
            currentPeriodStart: now, currentPeriodEnd: new Date(now.getTime() + 14*24*60*60*1000) }
  });
  console.log("Restaurant:", r.id);
  console.log("Subscription:", s.id);
  await prisma.$disconnect();
}

test();
---end file---

2. Run it:
   npx tsx test-create-restaurant.ts

STEP 2: VERIFY IN POSTGRESQL (Key Test)
================================================================================

Open terminal and run these commands:

1. List your test restaurant:

   psql $DATABASE_URL -c "
   SELECT id, name, status, 'trialEndsAt'
   FROM \"Restaurant\"
   WHERE status = 'TRIALING'
   ORDER BY \"createdAt\" DESC
   LIMIT 1;"

   EXPECTED:
   ✓ Shows your test restaurant
   ✓ status = TRIALING

   Save the ID if you don't have it: ____________________________


2. Verify subscription exists:

   psql $DATABASE_URL -c "
   SELECT
     ts.id,
     ts.status,
     ts.\"billingPeriod\",
     ts.\"autoRenew\",
     ts.\"nextBillingDate\",
     ts.\"lastBillingDate\",
     ts.\"paymentMethodId\"
   FROM \"TenantSubscription\" ts
   WHERE ts.\"restaurantId\" = 'YOUR_RESTAURANT_ID';"

   EXPECTED:
   ✓ Shows 1 subscription row
   ✓ status = TRIALING
   ✓ billingPeriod = MONTHLY
   ✓ autoRenew = true
   ✓ nextBillingDate = (empty/NULL)
   ✓ lastBillingDate = (empty/NULL)
   ✓ paymentMethodId = (empty/NULL)


3. Verify no billing data yet (expected):

   psql $DATABASE_URL -c "
   SELECT
     (SELECT COUNT(*) FROM \"PaymentMethod\" WHERE \"restaurantId\" = 'YOUR_RESTAURANT_ID') AS payment_methods,
     (SELECT COUNT(*) FROM \"PlatformInvoice\" WHERE \"restaurantId\" = 'YOUR_RESTAURANT_ID') AS invoices,
     (SELECT COUNT(*) FROM \"PlatformPayment\" WHERE \"restaurantId\" = 'YOUR_RESTAURANT_ID') AS payments;"

   EXPECTED:
   ✓ All three columns = 0 (empty is correct)


4. Verify relationships work:

   psql $DATABASE_URL -c "
   SELECT
     r.id AS restaurant_id,
     r.name,
     r.status,
     ts.id AS subscription_id,
     ts.status AS subscription_status
   FROM \"Restaurant\" r
   LEFT JOIN \"TenantSubscription\" ts ON r.id = ts.\"restaurantId\"
   WHERE r.id = 'YOUR_RESTAURANT_ID';"

   EXPECTED:
   ✓ Shows one row
   ✓ restaurant_id = subscription's restaurantId
   ✓ Both statuses = TRIALING
   ✓ LEFT JOIN shows the relationship works

================================================================================
TEST FAILED? TROUBLESHOOTING
================================================================================

Issue: "No restaurant found"
→ Run: psql $DATABASE_URL -c "SELECT COUNT(*) FROM \"Restaurant\";"
→ Check if migration applied: npx prisma migrate status
→ Verify app started correctly

Issue: "No subscription found"
→ Check Restaurant.currentPlanId is set
→ Verify trial plan exists: psql $DATABASE_URL -c "SELECT * FROM \"SubscriptionPlan\" WHERE code='trial';"
→ Check auth.ts creates subscription (should be automatic)

Issue: "Wrong number of billing tables"
→ Run: npx prisma migrate status
→ Verify all 3 tables exist: psql $DATABASE_URL -c "\dt Platform*"
→ Re-run migration if needed

Issue: "NULL values where expecting data"
→ This is often CORRECT for a new trial (except autoRenew should be true)
→ nextBillingDate/lastBillingDate/paymentMethodId being NULL is expected for new trial

Issue: "Foreign key constraint error"
→ Check parent records exist first
→ Verify no orphaned records:
  psql $DATABASE_URL -c "SELECT * FROM information_schema.referential_constraints WHERE table_name='TenantSubscription';"
